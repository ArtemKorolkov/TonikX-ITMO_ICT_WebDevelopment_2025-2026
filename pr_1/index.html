    Практическое задание 2.1: Создание модели данных об автовладельцах¶

    Цель: Создать модели данных в соответствии с требованиями.

Модели в файле automobiles/models.py:

from django.db import models

class Owner(models.Model):
    id = models.AutoField(primary_key=True)
    first_name = models.CharField(max_length=50, verbose_name="Имя", null=True, blank=True)
    last_name = models.CharField(max_length=50, verbose_name="Фамилия", null=True, blank=True)
    birth_date = models.DateField(verbose_name="Дата рождения", null=True, blank=True)
    phone = models.CharField(max_length=20, verbose_name="Телефон", null=True, blank=True)
    passport_number = models.CharField(max_length=20, verbose_name="Номер паспорта", null=True, blank=True)
    address = models.TextField(verbose_name="Домашний адрес", null=True, blank=True, max_length=500)
    nationality = models.CharField(max_length=10, choices=[
        ('RU', 'Российская Федерация'),
        ('BY', 'Республика Беларусь'),
        ('KZ', 'Республика Казахстан'),
        ('UA', 'Украина'),
        ('AM', 'Армения'),
        ('OTHER', 'Другая'),
    ], verbose_name="Национальность", null=True, blank=True)

    class Meta:
        db_table = 'owner'
        verbose_name = 'Владелец'
        verbose_name_plural = 'Владельцы'

    def __str__(self):
        return f"{self.first_name} {self.last_name}" if self.first_name and self.last_name else f"Владелец {self.id}"

class Car(models.Model):
    id = models.AutoField(primary_key=True, verbose_name="ID автомобиля")
    state_number = models.IntegerField(verbose_name="Гос. удостоверение")
    brand = models.CharField(max_length=30, verbose_name="Марка")
    model = models.CharField(max_length=20, verbose_name="Модель")
    color = models.CharField(max_length=30, choices=[
        ('RED', 'Красный'),
        ('BLUE', 'Синий'),
        ('GREEN', 'Зеленый'),
        ('BLACK', 'Черный'),
        ('WHITE', 'Белый'),
        ('SILVER', 'Серебристый'),
        ('GRAY', 'Серый'),
    ], null=True, blank=True, verbose_name="Цвет")

    class Meta:
        db_table = 'car'
        verbose_name = 'Автомобиль'
        verbose_name_plural = 'Автомобили'

    def __str__(self):
        return f"{self.brand} {self.model} ({self.state_number})"

class Ownership(models.Model):
    owner = models.ForeignKey(Owner, on_delete=models.CASCADE, verbose_name="Владелец", related_name='ownerships')
    car = models.ForeignKey(Car, on_delete=models.CASCADE, verbose_name="Автомобиль", related_name='ownerships')
    start_date = models.DateTimeField(verbose_name="Дата начала владения")
    end_date = models.DateTimeField(null=True, blank=True, verbose_name="Дата окончания владения")

    class Meta:
        db_table = 'ownership'
        verbose_name = 'Владение'
        verbose_name_plural = 'Владения'
        unique_together = ['owner', 'car', 'start_date']
        ordering = ['-start_date']

    def __str__(self):
        return f"{self.owner} - {self.car} ({self.start_date.date()})"

class DriverLicense(models.Model):
    id = models.AutoField(primary_key=True, verbose_name="ID удостоверения")
    owner = models.ForeignKey(Owner, on_delete=models.CASCADE, verbose_name="Владелец", related_name='licenses')
    license_number = models.CharField(max_length=10, verbose_name="Номер удостоверения")
    license_type = models.CharField(max_length=10, choices=[
        ('A', 'Категория A'),
        ('B', 'Категория B'),
        ('C', 'Категория C'),
        ('D', 'Категория D'),
        ('E', 'Категория E'),
    ], verbose_name="Тип удостоверения")
    issue_date = models.DateTimeField(verbose_name="Дата выдачи")

    class Meta:
        db_table = 'driver_license'
        verbose_name = 'Водительское удостоверение'
        verbose_name_plural = 'Водительские удостоверения'
        unique_together = ['license_number']

    def __str__(self):
        return f"Удостоверение {self.license_number} ({self.owner})"

Практическое задание 4: Создание контроллера и шаблона¶

Цель: Создать представление для отображения данных о владельце и соответствующий шаблон.

Контроллер в automobiles/views.py:

from django.shortcuts import render
from django.http import Http404
from .models import Owner

def owner_detail(request, owner_id):
    try:
        owner = Owner.objects.get(pk=owner_id)
    except Owner.DoesNotExist:
        raise Http404("Владелец не найден")
    return render(request, 'owner.html', {'owner': owner})
