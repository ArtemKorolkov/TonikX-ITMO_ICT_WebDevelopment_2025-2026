Отчет по практической работе №3.1¶

Django Web Framework. Запросы и их выполнение¶

Цель: Освоить выполнение запросов к базе данных в Django ORM, включая создание объектов, фильтрацию, агрегацию и сортировку данных.

Теоретическая основа¶

  • Django ORM - объектно-реляционное отображение для работы с базой данных
  • QuerySet API - интерфейс для построения запросов к базе данных
  • Методы фильтрации - filter(), exclude(), get()
  • Агрегирующие функции - Count(), Max(), Min(), Avg()
  • Сортировка данных - order_by(), distinct()

# queries.py
from django.db.models import Q
from autoapp.models import Owner, Car, DriverLicense, Ownership

def execute_queries():
    print("=== ЗАПРОСЫ НА ФИЛЬТРАЦИЮ ===\n")

    # 1. Все машины марки "Toyota"
    print("1. Все машины марки 'Toyota':")
    toyota_cars = Car.objects.filter(brand='Toyota')
    for car in toyota_cars:
        print(f"  - {car.brand} {car.model} ({car.color})")

    # 2. Все водители с именем "Олег"
    print("\n2. Все водители с именем 'Олег':")
    oleg_owners = Owner.objects.filter(name__contains='Олег')
    for owner in oleg_owners:
        print(f"  - {owner.name}")

    # 3. Получение удостоверения по id владельца
    print("\n3. Получение удостоверения по id владельца:")
    random_owner = Owner.objects.first()
    if random_owner:
        license = DriverLicense.objects.get(owner=random_owner)
        print(f"  Владелец: {random_owner.name}")
        print(f"  Удостоверение: {license.number}, выдано: {license.issued_date}")

    # 4. Все владельцы красных машин
    print("\n4. Все владельцы красных машин:")
    red_car_owners = Owner.objects.filter(
        ownerships__car__color='красный'
    ).distinct()
    for owner in red_car_owners:
        print(f"  - {owner.name}")

    # 5. Все владельцы, чей год владения машиной начинается с 2020
    print("\n5. Владельцы, начавшие владение с 2020 года:")
    owners_2020 = Owner.objects.filter(
        ownerships__start_date__year__gte=2020
    ).distinct()
    for owner in owners_2020:
        print(f"  - {owner.name}")

    return {
        'toyota_cars': toyota_cars,
        'oleg_owners': oleg_owners,
        'red_car_owners': red_car_owners,
        'owners_2020': owners_2020
    }

Реализация запросов¶

# advanced_queries.py
from django.db.models import Count, Max, Min
from django.db.models.functions import ExtractYear
from autoapp.models import Owner, Car, DriverLicense, Ownership

def execute_advanced_queries():
    print("=== СЛОЖНЫЕ ЗАПРОСЫ С АГРЕГАЦИЕЙ ===\n")

    # 1. Дата выдачи самого старшего водительского удостоверения
    print("1. Самое старое водительское удостоверение:")
    oldest_license = DriverLicense.objects.earliest('issued_date')
    print(f"  Выдано: {oldest_license.issued_date}, владелец: {oldest_license.owner.name}")

    # 2. Самая поздняя дата владения машиной
    print("\n2. Самая поздняя дата начала владения:")
    latest_ownership = Ownership.objects.latest('start_date')
    print(f"  Дата: {latest_ownership.start_date}, владелец: {latest_ownership.owner.name}, автомобиль: {latest_ownership.car.brand}")

    # 3. Количество машин для каждого водителя
    print("\n3. Количество машин у каждого владельца:")
    owners_car_count = Owner.objects.annotate(
        car_count=Count('ownerships')
    ).order_by('-car_count')

    for owner in owners_car_count:
        print(f"  {owner.name}: {owner.car_count} машин(ы)")

    # 4. Количество машин каждой марки
    print("\n4. Количество машин каждой марки:")
    cars_by_brand = Car.objects.values('brand').annotate(
        count=Count('id')
    ).order_by('-count')

    for brand in cars_by_brand:
        print(f"  {brand['brand']}: {brand['count']} машин(ы)")

    # 5. Сортировка владельцев по дате выдачи удостоверения
    print("\n5. Владельцы, отсортированные по дате выдачи удостоверения:")
    owners_sorted = Owner.objects.select_related('driverlicense').order_by(
        'driverlicense__issued_date'
    ).distinct()

    for owner in owners_sorted:
        license = owner.driverlicense
        print(f"  {owner.name}: удостоверение выдано {license.issued_date}")

    return {
        'oldest_license': oldest_license,
        'latest_ownership': latest_ownership,
        'owners_car_count': owners_car_count,
        'cars_by_brand': cars_by_brand,
        'owners_sorted': owners_sorted
    }

Модели данных¶

# models.py
from django.db import models

class Owner(models.Model):
    name = models.CharField(max_length=100, verbose_name='ФИО')
    birth_date = models.DateField(verbose_name='Дата рождения')

    def __str__(self):
        return self.name

    class Meta:
        verbose_name = 'Владелец'
        verbose_name_plural = 'Владельцы'

class Car(models.Model):
    brand = models.CharField(max_length=50, verbose_name='Марка')
    model = models.CharField(max_length=50, verbose_name='Модель')
    color = models.CharField(max_length=30, verbose_name='Цвет')
    year = models.IntegerField(verbose_name='Год выпуска')

    def __str__(self):
        return f"{self.brand} {self.model} ({self.year})"

    class Meta:
        verbose_name = 'Автомобиль'
        verbose_name_plural = 'Автомобили'

class DriverLicense(models.Model):
    LICENSE_TYPES = [
        ('A', 'Мотоциклы'),
        ('B', 'Легковые автомобили'),
        ('C', 'Грузовые автомобили'),
        ('D', 'Автобусы'),
    ]

    owner = models.OneToOneField(
        Owner, 
        on_delete=models.CASCADE, 
        verbose_name='Владелец',
        related_name='driverlicense'
    )
    number = models.CharField(max_length=10, verbose_name='Номер удостоверения')
    type = models.CharField(max_length=1, choices=LICENSE_TYPES, verbose_name='Тип')
    issued_date = models.DateField(verbose_name='Дата выдачи')

    def __str__(self):
        return f"{self.number} ({self.owner.name})"

    class Meta:
        verbose_name = 'Водительское удостоверение'
        verbose_name_plural = 'Водительские удостоверения'

class Ownership(models.Model):
    owner = models.ForeignKey(
        Owner, 
        on_delete=models.CASCADE, 
        verbose_name='Владелец',
        related_name='ownerships'
    )
    car = models.ForeignKey(
        Car, 
        on_delete=models.CASCADE, 
        verbose_name='Автомобиль',
        related_name='ownerships'
    )
    start_date = models.DateField(verbose_name='Дата начала владения')
    end_date = models.DateField(
        null=True, 
        blank=True, 
        verbose_name='Дата окончания владения'
    )

    def __str__(self):
        return f"{self.owner.name} - {self.car.brand} {self.car.model}"

    class Meta:
        verbose_name = 'Владение автомобилем'
        verbose_name_plural = 'Владения автомобилями'
